# Slerp between the two quaternions
def slerp(q1, q2, t):
    q1 = q1 / np.linalg.norm(q1)
    q2 = q2 / np.linalg.norm(q2)
    dot = np.clip(np.dot(q1, q2), -1.0, 1.0)
    if dot < 0:
        q2 = -q2
        dot = -dot
    if dot > 0.9999:
        return (q1 + t * (q2 - q1)) / np.linalg.norm(q1 + t * (q2 - q1))

    theta = theta_0 * t
    s1 = np.cos(theta) - dot * np.sin(theta) / np.sin(theta_0)
    s2 = np.sin(theta) / np.sin(theta_0)
    return s1 * q1 + s2 * q2

import numpy as np

def slerp(q1, q2, t):
    """Spherical linear interpolation between two quaternions [w,x,y,z]."""
    q1 = q1 / np.linalg.norm(q1)
    q2 = q2 / np.linalg.norm(q2)
    dot = np.dot(q1, q2)
    if dot < 0:
        q2 = -q2
        dot = -dot
    if dot > 0.9995:
        result = q1 + t * (q2 - q1)
        return result / np.linalg.norm(result)
    theta = np.arccos(np.clip(dot, -1, 1))
    sin_theta = np.sin(theta)
    w1 = np.sin((1 - t) * theta) / sin_theta
    w2 = np.sin(t * theta) / sin_theta
    return w1 * q1 + w2 * q2
